# Luca Demo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn this repo (branch `luca`) into a phone-first interactive dark-romance demo: text chat with Luca on DeepSeek, plus live PopVid video calls he can start himself.

**Architecture:** Express 5 serves the API and the built Vite/React app. One persona module feeds both the DeepSeek chat prompt and the PopVid video prompts. Chat history lives in the browser's `localStorage`; the server is stateless apart from an in-memory map of live PopVid sessions.

**Tech Stack:** Node 22 (ESM), Express 5, React 19, Vite 6, `node:test`, DeepSeek Chat Completions (OpenAI-compatible), PopVid Realtime API v1 (`r2-realtime-v1`).

**Spec:** `docs/superpowers/specs/2026-10-09-luca-demo-design.md`

## Global Constraints

- Branch: `luca`. Never commit `.env`.
- API keys (`POPVID_API_KEY`, `DEEPSEEK_API_KEY`) are read only on the server and never sent to the browser.
- PopVid: model `r2-realtime-v1`; `character.prompt` and `scene.prompt` each ≤ 4000 characters; `history` ≤ 40 messages, roles `user` / `character`, each `content` ≤ 2000 characters; seed image 576×768 at `/luca.jpg`; a call lasts at most 300 000 ms.
- DeepSeek: base URL default `https://api.deepseek.com`, model default `deepseek-chat`, JSON output (`response_format: { type: "json_object" }`), 8 000 ms timeout, one retry, at most 30 history messages sent.
- He calls on his own at most once per conversation; the server forces the call on the 6th user message if the model has not called yet.
- Content stays PG-13. When sincerely asked, Luca must not claim to be human.
- Crisis text (self-harm / suicide) never reaches DeepSeek or PopVid; the user gets `CRISIS_REPLY` (988 + Crisis Text Line).
- Message shape everywhere: `{ id, role: "user" | "him", text, kind: "text" | "call" | "notice", ts, durationMs?, retryable? }`. Only `kind: "text"` is ever sent to a model.
- Copy is English. UI is phone-first (portrait); it must also look fine on desktop (centered column, max width 480px).
- Do not wrap the React root in `StrictMode` (it would open two PopVid sessions per call).
- Tests: `npm test` runs `node --test server/*.test.js src/lib/*.test.js`; output must be clean.
- Commit messages: one plain English sentence ending with a period, like the existing history (e.g. `Add the Luca persona and crisis detection.`).

```js
export const POPVID_HISTORY_LIMIT = 40;
const MAX_CONTENT_CHARS = 2000;

function textMessages(messages) {
  return (Array.isArray(messages) ? messages : []).filter(
    (m) =>
      m &&
      (m.kind ?? "text") === "text" &&
      (m.role === "user" || m.role === "him") &&
      typeof m.text === "string" &&
      m.text.trim(),
  );
}

export function toPopvidHistory(messages) {
  return textMessages(messages)
    .slice(-POPVID_HISTORY_LIMIT)
    .map((m) => ({
      role: m.role === "him" ? "character" : "user",
      content: m.text.trim().slice(0, MAX_CONTENT_CHARS),
    }));
}

export function toChatMessages(messages, limit = 30) {
  return textMessages(messages)
    .slice(-limit)
    .map((m) => ({
      role: m.role === "him" ? "assistant" : "user",
      content: m.text.trim().slice(0, MAX_CONTENT_CHARS),
    }));
}
```

```js
import { chatSystemPrompt } from "./persona.js";
import { formatDuration } from "./format.js";
import { toChatMessages } from "./history.js";

export const FORCE_CALL_AT = 6;
export const CHAT_HISTORY_LIMIT = 30;
export const CHAT_TIMEOUT_MS = 8000;
export const FALLBACK_REPLY = "Give me a second.";

const fallback = () => ({ reply: FALLBACK_REPLY, call: false, quickReplies: [], retryable: true });

export function countUserTurns(messages) {
  return (Array.isArray(messages) ? messages : []).filter(
    (m) => m?.role === "user" && (m.kind ?? "text") === "text",
  ).length;
}

export function stageNote(event, durationMs) {
  if (event === "call_ended") {
    return `[The video call just ended after ${formatDuration(durationMs)}. Text her first, picking up from the call.]`;
  }
  if (event === "call_dropped") return "[The video call cut out unexpectedly. Text her about it, in character.]";
  if (event === "call_declined") return "[She declined your video call. Text her, unbothered but noticing.]";
  return null;
}

export function parseModelOutput(raw) {
  const text = String(raw ?? "").trim();
  let data = null;
  try {
    data = JSON.parse(text);
  } catch {
    data = null;
  }
  if (!data || typeof data !== "object") {
    return text ? { reply: text, call: false, quickReplies: [] } : fallback();
  }
  const reply = typeof data.reply === "string" ? data.reply.trim() : "";
  if (!reply) return fallback();
  const quickReplies = Array.isArray(data.quick_replies)
    ? data.quick_replies
        .filter((q) => typeof q === "string" && q.trim())
        .map((q) => q.trim())
        .slice(0, 3)
    : [];
  return { reply, call: data.call === true, quickReplies };
}

export function decideCall({ modelWantsCall, userTurns, calledOnce, event }) {
  if (calledOnce || event) return false;
  return Boolean(modelWantsCall) || userTurns >= FORCE_CALL_AT;
}

async function requestCompletion({ body, apiKey, baseUrl, fetchImpl, timeoutMs }) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  try {
    const res = await fetchImpl(`${baseUrl}/chat/completions`, {
      method: "POST",
      headers: { Authorization: `Bearer ${apiKey}`, "Content-Type": "application/json" },
      body: JSON.stringify(body),
      signal: controller.signal,
    });
    if (!res.ok) throw new Error(`DeepSeek responded ${res.status}`);
    const data = await res.json();
    return data?.choices?.[0]?.message?.content ?? "";
  } finally {
    clearTimeout(timer);
  }
}

export async function generateReply({
  messages,
  event = null,
  durationMs = 0,
  calledOnce = false,
  fetchImpl = fetch,
  env = process.env,
  timeoutMs = CHAT_TIMEOUT_MS,
}) {
  const apiKey = env.DEEPSEEK_API_KEY;
  if (!apiKey) return fallback();

  const baseUrl = (env.DEEPSEEK_BASE_URL || "https://api.deepseek.com").replace(/\/$/, "");
  const chat = toChatMessages(messages, CHAT_HISTORY_LIMIT);
  const note = stageNote(event, durationMs);
  if (note) chat.push({ role: "user", content: note });

  const body = {
    model: env.DEEPSEEK_MODEL || "deepseek-chat",
    messages: [{ role: "system", content: chatSystemPrompt() }, ...chat],
    response_format: { type: "json_object" },
    temperature: 1.1,
    max_tokens: 300,
  };

  let raw = null;
  for (let attempt = 0; attempt < 2 && raw === null; attempt += 1) {
    try {
      raw = await requestCompletion({ body, apiKey, baseUrl, fetchImpl, timeoutMs });
    } catch (err) {
      console.warn("[chat] DeepSeek attempt failed:", err.message);
    }
  }
  if (raw === null) return fallback();

  const parsed = parseModelOutput(raw);
  if (parsed.retryable) return parsed;
  return {
    ...parsed,
    call: decideCall({ modelWantsCall: parsed.call, userTurns: countUserTurns(messages), calledOnce, event }),
  };
}
```

```js
import { PERSONA, videoCharacterPrompt, videoScenePrompt } from "./persona.js";
import { toPopvidHistory } from "./history.js";

const BASE = process.env.POPVID_BASE_URL || "https://popvid.ai/api/public/v1";
const KEY = process.env.POPVID_API_KEY;
const PUBLIC_BASE = (process.env.PUBLIC_BASE_URL || "").replace(/\/$/, "");
const SEED_IMAGE_URL = (process.env.SEED_IMAGE_URL || "").trim();
const VOICE_REF_URL = (process.env.VOICE_REF_URL || "").trim();
const MAX_CONCURRENT = Number(process.env.MAX_CONCURRENT_SESSIONS || 3);

export const MAX_CALL_MS = 300_000;
const LEASE_MS = 25_000;
const BUSY_RETRY_MS = 5_000;

const calls = new Map();

function publicOrigin(req) {
  if (PUBLIC_BASE) return PUBLIC_BASE;
  const xfProto = String(req?.headers?.["x-forwarded-proto"] || "").split(",")[0].trim();
  const xfHost = String(req?.headers?.["x-forwarded-host"] || "").split(",")[0].trim();
  const host = xfHost || req?.headers?.host || "";
  const proto = xfProto || (req?.protocol === "https" || req?.secure ? "https" : "http");
  if (!host || /localhost|127\.0\.0\.1/i.test(host)) return "";
  return `${proto}://${host}`;
}

function seedUrl(req) {
  if (SEED_IMAGE_URL) return SEED_IMAGE_URL;
  const origin = publicOrigin(req);
  return origin ? `${origin}${PERSONA.photo}` : null;
}

function authHeaders() {
  return { Authorization: `Bearer ${KEY}`, "Content-Type": "application/json" };
}

export function buildSessionBody({ messages, seedImageUrl, voiceRefUrl }) {
  const character = { name: PERSONA.name, prompt: videoCharacterPrompt() };
  if (voiceRefUrl) character.voice_ref_url = voiceRefUrl;
  const body = {
    model: "r2-realtime-v1",
    character,
    scene: { prompt: videoScenePrompt() },
    language: "en",
    history: toPopvidHistory(messages),
    limits: { max_turns: 200, turn_rate_per_min: 20 },
    credentials_ttl_ms: 600_000,
    metadata: { product: "luca" },
  };
  if (seedImageUrl) body.seed_image_url = seedImageUrl;
  return body;
}

async function postConnection(body) {
  const res = await fetch(`${BASE}/connections`, {
    method: "POST",
    headers: authHeaders(),
    body: JSON.stringify(body),
  });
  const data = await res.json().catch(() => ({}));
  return { ok: res.ok, status: res.status, data };
}

export async function createCall({ messages, req }) {
  if (!KEY) {
    return { ok: false, status: 503, error: { code: "misconfigured", message: "Video is not configured." } };
  }
  if (calls.size >= MAX_CONCURRENT) {
    return { ok: false, status: 429, error: { code: "busy", retry_after_ms: BUSY_RETRY_MS } };
  }

  let result;
  try {
    result = await postConnection(buildSessionBody({ messages, seedImageUrl: seedUrl(req), voiceRefUrl: VOICE_REF_URL }));
    const code = result.data?.error?.code;
    if (!result.ok && !["no_capacity", "unauthorized", "budget_exhausted"].includes(code)) {
      console.warn("[call] retrying without seed after", result.status, code);
      result = await postConnection(buildSessionBody({ messages, seedImageUrl: null, voiceRefUrl: VOICE_REF_URL }));
    }
  } catch (err) {
    console.error("[call] PopVid unreachable", err);
    return { ok: false, status: 502, error: { code: "upstream_unreachable", message: "The call did not connect." } };
  }

  if (!result.ok) {
    const raw = result.data?.error || {};
    if (raw.code === "no_capacity") {
      return { ok: false, status: 429, error: { code: "busy", retry_after_ms: raw.retry_after_ms || BUSY_RETRY_MS } };
    }
    console.error("[call] rejected", result.status, raw);
    return {
      ok: false,
      status: result.status || 502,
      error: { code: raw.code || "upstream", message: raw.message || "The call did not connect." },
    };
  }

  const { session, credentials } = result.data || {};
  if (!session?.session_id || !credentials) {
    return { ok: false, status: 502, error: { code: "bad_payload", message: "The call did not connect." } };
  }

  const now = Date.now();
  calls.set(session.session_id, { createdAt: now, lastLease: now });
  return {
    ok: true,
    status: 201,
    data: {
      credentials,
      session: {
        session_id: session.session_id,
        reservation_expires_at_ms: session.reservation_expires_at_ms,
      },
    },
  };
}

export function renewLease(id) {
  const row = calls.get(id);
  if (!row) return false;
  row.lastLease = Date.now();
  return true;
}

export async function closeCall(id) {
  if (!id) return;
  calls.delete(id);
  if (!KEY) return;
  try {
    await fetch(`${BASE}/sessions/${encodeURIComponent(id)}`, { method: "DELETE", headers: authHeaders() });
  } catch (err) {
    console.warn("[call] close failed", id, err.message);
  }
}

export function sweepCalls(now = Date.now()) {
  for (const [id, row] of calls) {
    if (now - row.lastLease > LEASE_MS || now - row.createdAt > MAX_CALL_MS + 15_000) {
      closeCall(id);
    }
  }
}

export function activeCalls() {
  return calls.size;
}
```

```jsx
import { useEffect, useRef, useState } from "react";
import { R2Client } from "../lib/r2.js";
import { endCall, heartbeatCall, startCall } from "../lib/api.js";
import { detectCrisis } from "../../server/safety.js";
import { formatDuration } from "../../server/format.js";

const MAX_CALL_MS = 300_000;
const BUSY_GIVE_UP_MS = 30_000;
const WARN_MS = 30_000;
const GOODBYE_MS = 20_000;
const GOODBYE_LINE = "The user has to go in a moment.";
const ACTIONS = [
  { label: "smile", line: "The user smiles at you." },
  { label: "look away", line: "The user looks away, flustered." },
  { label: "bite your lip", line: "The user bites her lip." },
];

function endReason(reason) {
  if (reason === "budget_exhausted") return "time_up";
  if (reason === "client_closed" || reason === "server_closed") return "hangup";
  return "dropped";
}

export default function Call({ persona, messages, onLine, onEnd }) {
  const videoRef = useRef(null);
  const clientRef = useRef(null);
  const sessionRef = useRef(null);
  const startedAtRef = useRef(0);
  const billedRef = useRef(0);
  const goodbyeRef = useRef(false);
  const endedRef = useRef(false);
  const onLineRef = useRef(onLine);
  const onEndRef = useRef(onEnd);
  onLineRef.current = onLine;
  onEndRef.current = onEnd;

  const [phase, setPhase] = useState("connecting");
  const [caption, setCaption] = useState("");
  const [idle, setIdle] = useState(true);
  const [elapsed, setElapsed] = useState(0);
  const [needsTap, setNeedsTap] = useState(false);
  const [draft, setDraft] = useState("");

  function finish(reason) {
    if (endedRef.current) return;
    endedRef.current = true;
    const durationMs = startedAtRef.current ? Date.now() - startedAtRef.current : 0;
    clientRef.current?.close();
    endCall(sessionRef.current);
    onEndRef.current({ reason, durationMs });
  }

  function handleEvent(msg) {
    const d = msg.data || {};
    if (msg.type === "media.clip") {
      if (!startedAtRef.current) {
        startedAtRef.current = Date.now();
        setPhase("live");
      }
      setIdle(d.kind === "idle");
    }
    if (msg.type === "turn.text" && d.text) {
      setCaption(d.text);
      onLineRef.current("him", d.text);
    }
    if (msg.type === "usage.tick" && typeof d.billed_ms === "number") {
      billedRef.current = d.billed_ms;
    }
  }

  useEffect(() => {
    let cancelled = false;
    const firstTry = Date.now();

    function open({ credentials, session }) {
      sessionRef.current = session.session_id;
      const client = new R2Client({
        credentials,
        remoteVideo: videoRef.current,
        onRemoteStream: () => {
          videoRef.current?.play().catch(() => setNeedsTap(true));
        },
        onEvent: handleEvent,
        onError: (err) => {
          if (err?.code === "content_rejected") setCaption("Careful.");
        },
        onEnded: (data) => finish(endReason(data?.reason)),
      });
      clientRef.current = client;
      client.start();
    }

    async function connect() {
      while (!cancelled) {
        const res = await startCall(messages);
        if (cancelled) {
          if (res.ok) endCall(res.data.session.session_id);
          return;
        }
        if (res.ok) {
          open(res.data);
          return;
        }
        const error = res.data?.error || {};
        if (error.code === "busy" && Date.now() - firstTry < BUSY_GIVE_UP_MS) {
          setPhase("busy");
          await new Promise((resolve) => setTimeout(resolve, error.retry_after_ms || 5000));
          continue;
        }
        finish(error.code === "busy" ? "busy" : "dropped");
        return;
      }
    }

    connect();
    const beat = setInterval(() => heartbeatCall(sessionRef.current), 10_000);

    return () => {
      cancelled = true;
      clearInterval(beat);
      if (!endedRef.current) {
        endedRef.current = true;
        clientRef.current?.close();
        endCall(sessionRef.current);
      }
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  useEffect(() => {
    if (phase !== "live") return undefined;
    const tick = setInterval(() => {
      const ms = Math.max(Date.now() - startedAtRef.current, billedRef.current);
      setElapsed(ms);
      if (MAX_CALL_MS - ms <= GOODBYE_MS && !goodbyeRef.current) {
        goodbyeRef.current = true;
        clientRef.current?.say(GOODBYE_LINE);
      }
    }, 1000);
    return () => clearInterval(tick);
  }, [phase]);

  function send(text) {
    const value = text.trim();
    if (!value || phase !== "live") return;
    if (detectCrisis(value)) {
      finish("crisis");
      return;
    }
    clientRef.current?.say(value);
    onLineRef.current("user", value);
    setDraft("");
  }

  function act(action) {
    if (phase !== "live") return;
    clientRef.current?.say(action.line);
    onLineRef.current("user", `*${action.label}*`);
  }

  const remaining = MAX_CALL_MS - elapsed;
  const live = phase === "live";
  let clock = "";
  if (live) clock = remaining <= WARN_MS ? `${Math.max(0, Math.ceil(remaining / 1000))}s left` : formatDuration(elapsed);

  return (
    <div className="call">
      <video ref={videoRef} className="call-video" playsInline autoPlay />
      {!live && (
        <div className="call-cover">
          <img src={persona.photo} alt="" />
          <p>{phase === "busy" ? `${persona.shortName} is on another line…` : "Connecting…"}</p>
        </div>
      )}
      <div className={`call-vignette${idle && live ? " is-idle" : ""}`} />

      <header className="call-top">
        <span className="call-name">{persona.shortName}</span>
        <span className={`call-clock${live && remaining <= WARN_MS ? " is-warn" : ""}`}>{clock}</span>
        <span className="ai-badge">AI</span>
      </header>

      {needsTap && (
        <button
          type="button"
          className="call-unmute"
          onClick={() => {
            videoRef.current?.play().then(() => setNeedsTap(false)).catch(() => {});
          }}
        >
          Tap to hear him
        </button>
      )}

      {caption && <p className="call-caption">{caption}</p>}

      <footer className="call-bottom">
        <div className="call-actions">
          {ACTIONS.map((a) => (
            <button key={a.label} type="button" onClick={() => act(a)} disabled={!live}>
              *{a.label}*
            </button>
          ))}
        </div>
        <div className="call-row">
          <form
            className="call-input"
            onSubmit={(e) => {
              e.preventDefault();
              send(draft);
            }}
          >
            <input
              value={draft}
              onChange={(e) => setDraft(e.target.value)}
              placeholder={live ? "Say something…" : "Waiting for him…"}
              maxLength={500}
              disabled={!live}
            />
            <button type="submit" disabled={!live || !draft.trim()}>
              Send
            </button>
          </form>
          <button type="button" className="call-hangup" onClick={() => finish("hangup")} aria-label="End call">
            End
          </button>
        </div>
      </footer>
    </div>
  );
}
```

`stateRef` holds the latest state synchronously so that rapid `onLine` calls during a video call never drop a message.

```jsx
import { useEffect, useRef, useState } from "react";
import Gate from "./components/Gate.jsx";
import Chat from "./components/Chat.jsx";
import IncomingCall from "./components/IncomingCall.jsx";
import Call from "./components/Call.jsx";
import { fetchPersona, sendChat } from "./lib/api.js";
import { initialState, loadState, makeMessage, saveState } from "./lib/store.js";
import { CRISIS_REPLY, detectCrisis } from "../server/safety.js";
import { formatDuration } from "../server/format.js";

const RING_DELAY_MS = 1200;

export default function App() {
  const [persona, setPersona] = useState(null);
  const [state, setState] = useState(null);
  const [view, setView] = useState("chat");
  const [typing, setTyping] = useState(false);
  const stateRef = useRef(null);

  useEffect(() => {
    fetchPersona()
      .then((p) => {
        const s = loadState() || initialState(p);
        stateRef.current = s;
        setPersona(p);
        setState(s);
      })
      .catch(() => setPersona(false));
  }, []);

  useEffect(() => {
    if (state) saveState(state);
  }, [state]);

  function commit(next) {
    stateRef.current = next;
    setState(next);
  }

  function append(...msgs) {
    const s = stateRef.current;
    commit({ ...s, messages: [...s.messages, ...msgs] });
  }

  async function askHim(extra = {}) {
    setTyping(true);
    const before = stateRef.current;
    const res = await sendChat({ messages: before.messages, calledOnce: before.calledOnce, ...extra });
    setTyping(false);

    let fields = {};
    if (res.notice) fields = { kind: "notice" };
    else if (res.retryable) fields = { kind: "notice", retryable: true, retryWith: extra };

    const s = stateRef.current;
    const quickReplies = res.quickReplies?.length ? res.quickReplies : s.quickReplies;
    const willCall = Boolean(res.call) && !s.calledOnce;
    commit({
      ...s,
      calledOnce: s.calledOnce || willCall,
      quickReplies,
      messages: [...s.messages, makeMessage("him", res.reply, fields)],
    });
    if (willCall) {
      setTimeout(() => setView((v) => (v === "chat" ? "incoming" : v)), RING_DELAY_MS);
    }
  }

  function sendText(text) {
    const value = text.trim();
    if (!value) return;
    if (detectCrisis(value)) {
      append(makeMessage("user", value, { kind: "notice" }), makeMessage("him", CRISIS_REPLY, { kind: "notice" }));
      return;
    }
    append(makeMessage("user", value));
    askHim();
  }

  function resend(id) {
    const s = stateRef.current;
    const failed = s.messages.find((m) => m.id === id);
    commit({ ...s, messages: s.messages.filter((m) => m.id !== id) });
    askHim(failed?.retryWith || {});
  }

  function passGate() {
    commit({ ...stateRef.current, gatePassed: true });
  }

  function callNow() {
    commit({ ...stateRef.current, calledOnce: true });
    setView("call");
  }

  function decline() {
    setView("chat");
    askHim({ event: "call_declined" });
  }

  function onLine(role, text) {
    append(makeMessage(role, text));
  }

  function onEnd({ reason, durationMs }) {
    setView("chat");
    if (reason === "busy") {
      append(makeMessage("him", "I'll call you back."));
      return;
    }
    if (reason === "crisis") {
      append(makeMessage("him", CRISIS_REPLY, { kind: "notice" }));
      return;
    }
    append(makeMessage("him", `Video call · ${formatDuration(durationMs)}`, { kind: "call", durationMs }));
    askHim({ event: reason === "dropped" ? "call_dropped" : "call_ended", durationMs });
  }

  if (persona === false) {
    return <div className="screen-center">Couldn't reach Luca. Refresh to try again.</div>;
  }
  if (!persona || !state) return <div className="screen-center" />;
  if (!state.gatePassed) return <Gate persona={persona} onEnter={passGate} />;

  return (
    <>
      <Chat
        persona={persona}
        messages={state.messages}
        quickReplies={state.quickReplies}
        typing={typing}
        onSend={sendText}
        onCall={callNow}
        onResend={resend}
      />
      {view === "incoming" && <IncomingCall persona={persona} onAnswer={() => setView("call")} onDecline={decline} />}
      {view === "call" && <Call persona={persona} messages={state.messages} onLine={onLine} onEnd={onEnd} />}
    </>
  );
}
```
